## Review Focus

Situations the spec is silent on, most likely to bite first. Each has a test in the task named.

```swift
import CoreLocation
import Foundation

/// What is leaving near a point, computed by the server (GET /api/v1/nearby).
///
/// `stops` lists only the stops that `departures` reference; they become the
/// map pins. Lists are `[]` on the wire, never null, but a missing or null
/// list decodes as empty rather than failing the whole response.
struct NearbyResponse: Decodable, Equatable, Sendable {
    let stops: [DepartureStop]
    let departures: [NearbyDeparture]

    static let empty = NearbyResponse(stops: [], departures: [])

    init(stops: [DepartureStop], departures: [NearbyDeparture]) {
        self.stops = stops
        self.departures = departures
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        stops = try container.decodeIfPresent([DepartureStop].self, forKey: .stops) ?? []
        departures = try container.decodeIfPresent([NearbyDeparture].self, forKey: .departures) ?? []
    }

    private enum CodingKeys: String, CodingKey {
        case stops, departures
    }

    func stop(withId id: String) -> DepartureStop? {
        stops.first { $0.id == id }
    }
}

/// A stop the timeline uses, with its distance from the query point.
struct DepartureStop: Decodable, Identifiable, Hashable, Sendable {
    let id: String
    let name: String
    let lat: Double
    let lon: Double
    let distanceM: Int

    var coordinate: CLLocationCoordinate2D {
        CLLocationCoordinate2D(latitude: lat, longitude: lon)
    }

    enum CodingKeys: String, CodingKey {
        case id, name, lat, lon
        case distanceM = "distance_m"
    }
}

/// One timeline row: a route and direction, from the nearest stop that has
/// a departure in the window, with its next (up to three) times.
struct NearbyDeparture: Decodable, Identifiable, Hashable, Sendable {
    let route: String
    let agencyId: String
    let direction: Int
    /// The trip's headsign; the server omits it when none is known.
    let headsign: String?
    let stopId: String
    let distanceM: Int
    let times: [DepartureTime]

    /// One row per (route, agency, direction), so that is the identity.
    var id: String { "\(agencyId)|\(route)|\(direction)" }
    var routeRef: RouteRef { RouteRef(name: route, agencyId: agencyId) }
    var firstTime: DepartureTime? { times.first }

    /// What the row shows as the destination: the headsign, else the
    /// direction, so a row is never blank.
    var destination: String {
        if let headsign, !headsign.isEmpty { return headsign }
        return "Direction \(direction)"
    }

    enum CodingKeys: String, CodingKey {
        case route, direction, headsign, times
        case agencyId = "agency_id"
        case stopId = "stop_id"
        case distanceM = "distance_m"
    }
}

/// Where a departure time came from. Live and realtime are shown in the
/// brand colour; timetable-only times are muted.
enum DepartureSource: String, Decodable, Sendable {
    /// A vehicle is reporting for the trip (or a Luas forecast).
    case live
    /// A trip update exists but no vehicle reports.
    case realtime
    /// Scheduled time only.
    case timetable

    var isLive: Bool { self != .timetable }
}

/// One departure time of a trip at a stop.
struct DepartureTime: Decodable, Hashable, Sendable {
    let tripId: String
    let at: Date
    /// Seconds late (positive) or early.
    let delay: Int
    let source: DepartureSource
    /// A cancelled trip keeps its timetabled time.
    let cancelled: Bool

    init(tripId: String, at: Date, delay: Int, source: DepartureSource, cancelled: Bool) {
        self.tripId = tripId
        self.at = at
        self.delay = delay
        self.source = source
        self.cancelled = cancelled
    }

    /// An unknown source reads as timetable and a missing `cancelled` as
    /// false, so a newer server never blanks the timeline.
    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        tripId = try container.decode(String.self, forKey: .tripId)
        at = try container.decode(Date.self, forKey: .at)
        delay = try container.decodeIfPresent(Int.self, forKey: .delay) ?? 0
        let rawSource = try container.decodeIfPresent(String.self, forKey: .source) ?? ""
        source = DepartureSource(rawValue: rawSource) ?? .timetable
        cancelled = try container.decodeIfPresent(Bool.self, forKey: .cancelled) ?? false
    }

    private enum CodingKeys: String, CodingKey {
        case at, delay, source, cancelled
        case tripId = "trip_id"
    }
}
```

```swift
import CoreLocation
import Foundation

/// Stops and routes matching a query (GET /api/v1/search). Places come from
/// MapKit on the device, not from the server.
struct SearchResponse: Decodable, Equatable, Sendable {
    let stops: [StopSearchResult]
    let routes: [RouteSearchResult]

    static let empty = SearchResponse(stops: [], routes: [])

    init(stops: [StopSearchResult], routes: [RouteSearchResult]) {
        self.stops = stops
        self.routes = routes
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        stops = try container.decodeIfPresent([StopSearchResult].self, forKey: .stops) ?? []
        routes = try container.decodeIfPresent([RouteSearchResult].self, forKey: .routes) ?? []
    }

    private enum CodingKeys: String, CodingKey {
        case stops, routes
    }
}

/// A stop found by name. Selecting it flies the map there.
struct StopSearchResult: Decodable, Identifiable, Hashable, Sendable {
    let id: String
    let name: String
    let lat: Double
    let lon: Double
    /// Modes serving the stop; unknown mode names are dropped.
    let modes: [TransportMode]

    var coordinate: CLLocationCoordinate2D {
        CLLocationCoordinate2D(latitude: lat, longitude: lon)
    }

    init(id: String, name: String, lat: Double, lon: Double, modes: [TransportMode]) {
        self.id = id
        self.name = name
        self.lat = lat
        self.lon = lon
        self.modes = modes
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        id = try container.decode(String.self, forKey: .id)
        name = try container.decode(String.self, forKey: .name)
        lat = try container.decode(Double.self, forKey: .lat)
        lon = try container.decode(Double.self, forKey: .lon)
        let rawModes = try container.decodeIfPresent([String].self, forKey: .modes) ?? []
        modes = rawModes.compactMap(TransportMode.init(rawValue:))
    }

    private enum CodingKeys: String, CodingKey {
        case id, name, lat, lon, modes
    }
}

/// A route found by short name or headsign. Selecting it opens route focus.
struct RouteSearchResult: Decodable, Identifiable, Hashable, Sendable {
    let name: String
    let agencyId: String
    let agencyName: String?
    let mode: TransportMode
    let headsignOutbound: String?
    let headsignInbound: String?

    var id: String { "\(agencyId)|\(name)" }
    var ref: RouteRef { RouteRef(name: name, agencyId: agencyId) }

    /// Agency name for display, falling back to the agency ID.
    var operatorLabel: String {
        if let agencyName, !agencyName.isEmpty {
            return agencyName
        }
        return agencyId.isEmpty ? "Unknown operator" : "Operator \(agencyId)"
    }

    func headsign(for direction: Int) -> String? {
        let headsign = direction == 0 ? headsignOutbound : headsignInbound
        guard let headsign, !headsign.isEmpty else { return nil }
        return headsign
    }

    init(
        name: String,
        agencyId: String,
        agencyName: String? = nil,
        mode: TransportMode,
        headsignOutbound: String? = nil,
        headsignInbound: String? = nil
    ) {
        self.name = name
        self.agencyId = agencyId
        self.agencyName = agencyName
        self.mode = mode
        self.headsignOutbound = headsignOutbound
        self.headsignInbound = headsignInbound
    }
```

```swift
import Foundation
@testable import Turas

/// A clock tests move by hand. `advance(by:)` wakes every sleeper whose
/// deadline has passed, in deadline order, so a model's debounce and
/// refresh loops run deterministically.
///
/// A sleeper is registered when the sleeping task reaches `sleep(for:)`,
/// which is after the call that started it returns; wait for
/// `sleeperCount` before advancing.
final class ManualClock: @unchecked Sendable {
    private struct Sleeper {
        let id: UUID
        let wake: Date
        let continuation: CheckedContinuation<Void, Error>
    }

    /// Deadlines are reached by repeated additions; allow a rounding slop.
    private static let tolerance: TimeInterval = 1e-6

    private let lock = NSLock()
    private var current: Date
    private var sleepers: [Sleeper] = []

    init(now: Date = Date(timeIntervalSince1970: 1_791_630_000)) { // 2026-10-10T11:00:00Z
        current = now
    }

    var now: Date {
        lock.withLock { current }
    }

    /// Tasks waiting on this clock. A model that has finished a refresh is
    /// asleep until its next tick, so tests wait for this to settle.
    var sleeperCount: Int {
        lock.withLock { sleepers.count }
    }

    var appClock: AppClock {
        AppClock(now: { [self] in self.now }, sleep: { [self] in try await self.sleep(for: $0) })
    }

    func advance(by seconds: TimeInterval) {
        let due: [Sleeper] = lock.withLock {
            current = current.addingTimeInterval(seconds)
            let limit = current.addingTimeInterval(Self.tolerance)
            let due = sleepers.filter { $0.wake <= limit }.sorted { $0.wake < $1.wake }
            sleepers.removeAll { $0.wake <= limit }
            return due
        }
        for sleeper in due {
            sleeper.continuation.resume()
        }
    }

    func sleep(for duration: Duration) async throws {
        let id = UUID()
        let seconds = Self.seconds(duration)
        try await withTaskCancellationHandler {
            try await withCheckedThrowingContinuation { (continuation: CheckedContinuation<Void, Error>) in
                let immediate: Bool = lock.withLock {
                    if Task.isCancelled { return true }
                    let wake = current.addingTimeInterval(seconds)
                    guard wake > current.addingTimeInterval(Self.tolerance) else { return true }
                    sleepers.append(Sleeper(id: id, wake: wake, continuation: continuation))
                    return false
                }
                if immediate {
                    continuation.resume()
                }
            }
            try Task.checkCancellation()
        } onCancel: {
            let cancelled: Sleeper? = lock.withLock {
                guard let index = sleepers.firstIndex(where: { $0.id == id }) else { return nil }
                return sleepers.remove(at: index)
            }
            cancelled?.continuation.resume(throwing: CancellationError())
        }
    }

    static func seconds(_ duration: Duration) -> TimeInterval {
        let parts = duration.components
        return TimeInterval(parts.seconds) + TimeInterval(parts.attoseconds) / 1e18
    }
}
```

```swift
/// A scripted NearbySource. With a clock and a delay it sleeps on the test
/// clock before answering, so tests can interleave responses. The sleep is
/// detached, so cancelling the caller does not cut the answer short: tests
/// of the models' generation guards need the late answer to arrive.
actor FakeNearbySource: NearbySource {
    static func wait(_ delay: Duration, on clock: ManualClock) async {
        await Task.detached { try? await clock.sleep(for: delay) }.value
    }

    private let clock: ManualClock?
    private var responses: [TransportMode: NearbyResponse] = [.bus: SampleData.nearby(mode: .bus), .luas: SampleData.nearby(mode: .luas), .train: SampleData.nearby(mode: .train)]
    private var error: Error?
    private var delay: Duration?
    private(set) var calls: [NearbyCall] = [] {
        didSet { callCountBox.value = calls.count }
    }
    /// `calls.count`, readable without awaiting (for `waitUntil` closures).
    nonisolated var callCount: Int { callCountBox.value }
    private nonisolated let callCountBox = AtomicInt()

    init(clock: ManualClock? = nil) {
        self.clock = clock
    }

    func setResponse(_ response: NearbyResponse, for mode: TransportMode) { responses[mode] = response }
    func setError(_ error: Error?) { self.error = error }
    func setDelay(_ delay: Duration?) { self.delay = delay }

    func fetchNearby(lat: Double, lon: Double, mode: TransportMode) async throws -> NearbyResponse {
        calls.append(NearbyCall(lat: lat, lon: lon, mode: mode))
        if let delay, let clock {
            await Self.wait(delay, on: clock)
        }
        if let error { throw error }
        return responses[mode] ?? .empty
    }
}
```

```swift
    static func timeColor(for source: DepartureSource, cancelled: Bool) -> Color {
```

```swift
import CoreLocation
import Foundation
import os
import SwiftUI

/// Fetches what is leaving near the map centre for one mode, and keeps the
/// last good answer so the timeline never goes blank on a failed refresh.
///
/// The camera reports where it settled; about 0.6 s later (if it moved more
/// than 50 m from the last fetch) the rows are fetched, then refreshed
/// every 30 s while the app is active. A (centre, mode) change bumps
/// `generation`, and a response for an older generation is dropped.
@MainActor
final class NearbyModel: ObservableObject {
    /// What the timeline shows besides its rows.
    enum Status: Equatable {
        /// Nothing fetched yet for this mode; a request is pending or in flight.
        case loading
        /// Rows are current.
        case loaded
        /// The server answered with no departures.
        case empty
        /// The server is still loading its schedule (503); retried automatically.
        case starting
        /// The last refresh failed; `result` is the last good answer, if any.
        case failed(message: String)
    }

    static let settleDelay: Duration = .milliseconds(600)
    static let settleDistance: Double = 50
    static let refreshInterval: Duration = .seconds(30)

    @Published private(set) var result: NearbyResponse?
    @Published private(set) var lastUpdated: Date?
    @Published private(set) var status: Status = .loading
    @Published private(set) var mode: TransportMode

    private let source: NearbySource
    private let clock: AppClock
    /// The centre the current rows (or the in-flight request) are for.
    private(set) var fetchedCentre: CLLocationCoordinate2D?
    private var settleTask: Task<Void, Never>?
    private var refreshTask: Task<Void, Never>?
    private var generation = 0
    private var isActive = true

    private static let logger = Logger(subsystem: "com.turas", category: "NearbyModel")

    init(source: NearbySource, clock: AppClock = .system, mode: TransportMode = .bus) {
        self.source = source
        self.clock = clock
        self.mode = mode
    }

    var departures: [NearbyDeparture] { result?.departures ?? [] }
    var stops: [DepartureStop] { result?.stops ?? [] }
    /// Whether the settle timer is running (a fetch will follow).
    var hasPendingFetch: Bool { settleTask != nil }

    /// The first row at a stop; tapping a pin scrolls to it.
    func firstRow(atStop stopId: String) -> NearbyDeparture? {
        departures.first { $0.stopId == stopId }
    }

    // MARK: - Camera

    /// The map camera stopped at `centre`. Fetches 0.6 s later unless the
    /// centre is within 50 m of the last fetch. A newer settle replaces a
    /// pending one, so panning across town fetches only where it ends.
    func cameraSettled(at centre: CLLocationCoordinate2D) {
        settleTask?.cancel()
        settleTask = nil
        guard isActive else { return }
        if let fetchedCentre, Self.distance(fetchedCentre, centre) < Self.settleDistance {
            return
        }
        let clock = clock
        settleTask = Task { [weak self] in
            guard (try? await clock.sleep(for: Self.settleDelay)) != nil, let self, !Task.isCancelled else { return }
            self.settleTask = nil
            self.beginFetching(centre: centre)
        }
    }

    // MARK: - Mode, retry, lifecycle

    /// Switches mode: the old mode's rows go and the new mode is fetched now.
    func setMode(_ mode: TransportMode) {
        guard mode != self.mode else { return }
        self.mode = mode
        result = nil
        lastUpdated = nil
        status = .loading
        if let fetchedCentre {
            beginFetching(centre: fetchedCentre)
        }
    }

    /// Fetches again now (the Retry button).
    func retry() {
        guard let fetchedCentre else { return }
        beginFetching(centre: fetchedCentre)
    }

    func scenePhaseChanged(to phase: ScenePhase) {
        switch phase {
        case .active:
            resume()
        case .background:
            pause()
        default:
            break
        }
    }

    /// Stops the timer and the refresh loop (the app left the foreground,
    /// or route focus is showing).
    func pause() {
        isActive = false
        settleTask?.cancel()
        settleTask = nil
        refreshTask?.cancel()
        refreshTask = nil
    }

    /// Fetches now and restarts the 30 s refresh. The old rows and their
    /// timestamp stay until the fetch lands.
    func resume() {
        guard !isActive else { return }
        isActive = true
        if let fetchedCentre {
            beginFetching(centre: fetchedCentre)
        }
    }

    // MARK: - Fetching

    private func beginFetching(centre: CLLocationCoordinate2D) {
        fetchedCentre = centre
        generation += 1
        let generation = generation
        refreshTask?.cancel()
        guard isActive else { return }
        let clock = clock
        refreshTask = Task { [weak self] in
            while !Task.isCancelled {
                await self?.fetch(centre: centre, generation: generation)
                guard (try? await clock.sleep(for: Self.refreshInterval)) != nil else { return }
            }
        }
    }

    private func fetch(centre: CLLocationCoordinate2D, generation: Int) async {
        let mode = mode
        do {
            let response = try await source.fetchNearby(lat: centre.latitude, lon: centre.longitude, mode: mode)
            guard generation == self.generation else { return }
            result = response
            lastUpdated = clock.now()
            status = response.departures.isEmpty ? .empty : .loaded
        } catch {
            guard generation == self.generation, !ServerErrorText.isCancellation(error) else { return }
            Self.logger.error("Nearby fetch failed: \(error.localizedDescription)")
            switch error as? TurasServerError {
            case .httpError(statusCode: 503):
                status = .starting
            case .httpError(statusCode: 429):
                break // The next tick retries; the user never sees a rate limit.
            default:
                status = .failed(message: ServerErrorText.message(for: error))
            }
        }
    }

    nonisolated static func distance(_ a: CLLocationCoordinate2D, _ b: CLLocationCoordinate2D) -> Double {
        GeoUtils.straightLineDistance(lat1: a.latitude, lon1: a.longitude, lat2: b.latitude, lon2: b.longitude)
    }
}
```

```swift
/// Apple local search, limited to Ireland. Runs on the device; the server
/// never sees place queries.
struct MapKitPlaceSearch: PlaceSearching {
    /// A region covering the island; MapKit prefers results inside it.
    static let ireland = MKCoordinateRegion(
        center: CLLocationCoordinate2D(latitude: 53.4, longitude: -8.0),
        span: MKCoordinateSpan(latitudeDelta: 4.5, longitudeDelta: 6.0)
    )

    init() {}

    func searchPlaces(_ query: String) async throws -> [PlaceResult] {
        let request = MKLocalSearch.Request()
        request.naturalLanguageQuery = query
        request.region = Self.ireland
        request.resultTypes = [.pointOfInterest, .address]
        let response = try await MKLocalSearch(request: request).start()
        return response.mapItems.compactMap { item in
            guard let name = item.name, !name.isEmpty else { return nil }
            let coordinate = item.placemark.coordinate
            let subtitle = [item.placemark.thoroughfare, item.placemark.locality]
                .compactMap { $0 }
                .joined(separator: ", ")
            return PlaceResult(
                id: "\(name)|\(coordinate.latitude)|\(coordinate.longitude)",
                name: name,
                subtitle: subtitle.isEmpty ? nil : subtitle,
                lat: coordinate.latitude,
                lon: coordinate.longitude
            )
        }
    }
}
```

```swift
import Foundation
import os

/// One search box, three groups: stops and routes from the server, places
/// from MapKit. Typing is debounced 250 ms; the server and MapKit are
/// queried together and either may fail without hiding the other. A newer
/// query cancels the older one, and an older answer is never applied.
@MainActor
final class SearchModel: ObservableObject {
    struct Results: Equatable, Sendable {
        var stops: [StopSearchResult] = []
        var routes: [RouteSearchResult] = []
        var places: [PlaceResult] = []

        var isEmpty: Bool { stops.isEmpty && routes.isEmpty && places.isEmpty }

        static let none = Results()
    }

    static let debounce: Duration = .milliseconds(250)
    static let groupLimit = 8

    /// The text in the search field. Setting it schedules a search.
    @Published var query = "" {
        didSet { if query != oldValue { queryChanged() } }
    }
    @Published private(set) var results = Results.none
    /// True from the first keystroke until results (or nothing) arrive.
    @Published private(set) var isSearching = false
    @Published private(set) var mode: TransportMode

    private let source: SearchSource
    private let places: PlaceSearching
    private let clock: AppClock
    private var task: Task<Void, Never>?
    private var generation = 0

    private static let logger = Logger(subsystem: "com.turas", category: "SearchModel")

    init(source: SearchSource, places: PlaceSearching, clock: AppClock = .system, mode: TransportMode = .bus) {
        self.source = source
        self.places = places
        self.clock = clock
        self.mode = mode
    }

    /// The mode whose results come first. Re-runs the current query.
    func setMode(_ mode: TransportMode) {
        guard mode != self.mode else { return }
        self.mode = mode
        queryChanged()
    }

    /// Back to the timeline: no text, no results, nothing pending.
    func clear() {
        query = ""
        if results != .none || isSearching {
            // The didSet only fires on a change; an already-empty query
            // still needs the pending work dropped.
            queryChanged()
        }
    }

    // MARK: - Searching

    private func queryChanged() {
        task?.cancel()
        task = nil
        generation += 1
        let generation = generation
        let text = query.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !text.isEmpty else {
            results = .none
            isSearching = false
            return
        }
        isSearching = true
        let (source, places, clock, mode) = (source, places, clock, mode)
        task = Task { [weak self] in
            guard (try? await clock.sleep(for: Self.debounce)) != nil else { return }
            async let server: SearchResponse? = {
                do {
                    return try await source.search(query: text, mode: mode)
                } catch {
                    if !ServerErrorText.isCancellation(error) {
                        Self.logger.warning("Search failed: \(error.localizedDescription)")
                    }
                    return nil
                }
            }()
            async let found: [PlaceResult] = {
                do {
                    return try await places.searchPlaces(text)
                } catch {
                    Self.logger.warning("Place search failed: \(error.localizedDescription)")
                    return []
                }
            }()
            let (response, placeResults) = await (server, found)
            guard let self, !Task.isCancelled, self.generation == generation else { return }
            self.results = Results(
                stops: Array((response?.stops ?? []).prefix(Self.groupLimit)),
                routes: Array((response?.routes ?? []).prefix(Self.groupLimit)),
                places: Array(placeResults.filter(Self.isInIreland).prefix(Self.groupLimit))
            )
            self.isSearching = false
        }
    }

    /// MapKit can answer with places far outside the region it was given;
    /// those would fly the map off the transit network.
    nonisolated static func isInIreland(_ place: PlaceResult) -> Bool {
        (51.3...55.5).contains(place.lat) && (-10.8...(-5.3)).contains(place.lon)
    }
}
```

```swift
import CoreLocation
import Foundation

/// A stop to highlight in route focus: from a timeline pin, a route stop or
/// the map centre's nearest stop.
struct FocusStop: Hashable, Sendable {
    let id: String
    let name: String
    let lat: Double
    let lon: Double

    init(id: String, name: String, lat: Double, lon: Double) {
        self.id = id
        self.name = name
        self.lat = lat
        self.lon = lon
    }

    init(_ stop: DepartureStop) {
        self.init(id: stop.id, name: stop.name, lat: stop.lat, lon: stop.lon)
    }

    init(_ stop: RouteStop) {
        self.init(id: stop.id, name: stop.name, lat: stop.lat, lon: stop.lon)
    }

    var coordinate: CLLocationCoordinate2D {
        CLLocationCoordinate2D(latitude: lat, longitude: lon)
    }

    /// As a route stop (no sequence or offset known).
    var routeStop: RouteStop {
        RouteStop(id: id, name: name, lat: lat, lon: lon, seq: 0, t: 0)
    }
}

/// What route focus was opened for.
struct RouteFocusTarget: Hashable, Sendable {
    let route: RouteRef
    let mode: TransportMode
    let direction: Int
    /// Known from the timeline row or the search result; nil asks the
    /// route list and the bundled lookup.
    let headsign: String?
    /// The stop to highlight; nil (from a route search result) means the
    /// stop on the route nearest the map centre.
    let stop: FocusStop?
}

/// One route, direction and stop: the line (bundled shape), the stops, the
/// live vehicles on that direction and the next arrivals at the stop,
/// refreshed every 30 s while open. Flipping the direction keeps the
/// user near the same place: the nearest stop in the other direction.
@MainActor
final class RouteFocusModel: ObservableObject {
    static let refreshInterval: Duration = .seconds(30)
    /// How much of the line around the stop the camera fits, in metres.
    static let focusRadius: Double = 800

    let route: RouteRef
    let mode: TransportMode

    @Published private(set) var direction: Int
    @Published private(set) var headsign: String?
    @Published private(set) var stops: [RouteStop] = []
    @Published private(set) var shape: RouteShape?
    @Published private(set) var highlightedStop: RouteStop?
    @Published private(set) var vehicles: [Vehicle] = []
    @Published private(set) var arrivals: [StopArrival] = []
    @Published private(set) var lastUpdated: Date?
    @Published private(set) var errorMessage: String?
    @Published private(set) var isLoading = true

    private let source: FocusSource
    private let staticData: StaticDataStore
    private let shapes: RouteShapeService
    private let clock: AppClock
    private let headsignLookup: [String: String]
    private let centre: CLLocationCoordinate2D
    private let userLocation: CLLocationCoordinate2D?
    private let targetHeadsign: String?
    private var requestedStop: FocusStop?
    private var refreshTask: Task<Void, Never>?
    private var generation = 0
    private var isActive = true

    private static let logger = Logger(subsystem: "com.turas", category: "RouteFocusModel")

    init(
        target: RouteFocusTarget,
        centre: CLLocationCoordinate2D,
        userLocation: CLLocationCoordinate2D?,
        source: FocusSource,
        staticData: StaticDataStore,
        shapes: RouteShapeService,
        clock: AppClock = .system,
        headsignLookup: [String: String] = BundleLoader.loadStringLookup(resource: "headsign_lookup")
    ) {
        self.route = target.route
        self.mode = target.mode
        self.direction = target.direction
        self.headsign = target.headsign
        self.targetHeadsign = target.headsign
        self.requestedStop = target.stop
        self.centre = centre
        self.userLocation = userLocation
        self.source = source
        self.staticData = staticData
        self.shapes = shapes
        self.clock = clock
        self.headsignLookup = headsignLookup
    }

    /// Minutes to walk from the user's location to the highlighted stop.
    var walkingMinutes: Int? {
        guard let userLocation, let stop = highlightedStop else { return nil }
        let metres = GeoUtils.straightLineDistance(
            lat1: userLocation.latitude, lon1: userLocation.longitude, lat2: stop.lat, lon2: stop.lon
        )
        return GeoUtils.walkingMinutes(metres: metres)
    }

    /// Metres from the map centre to the highlighted stop (shown when the
    /// user's location is unknown).
    var distanceFromCentre: Int? {
        guard let stop = highlightedStop else { return nil }
        return Int(GeoUtils.straightLineDistance(lat1: centre.latitude, lon1: centre.longitude, lat2: stop.lat, lon2: stop.lon).rounded())
    }

    /// The camera for this focus: the stop and the nearby part of the line.
    var focusRegion: MKCoordinateRegion? {
        guard let stop = highlightedStop else { return nil }
        return Self.focusRegion(stop: stop, shape: shape)
    }

    // MARK: - Lifecycle

    /// Loads the stops and shape, picks the stop, then fetches and keeps
    /// refreshing every 30 s.
    func start() async {
        await load()
        startRefreshing()
    }

    func stop() {
        isActive = false
        refreshTask?.cancel()
        refreshTask = nil
    }

    func scenePhaseChanged(to phase: ScenePhase) {
        switch phase {
        case .active:
            guard !isActive else { return }
            isActive = true
            startRefreshing()
        case .background:
            isActive = false
            refreshTask?.cancel()
            refreshTask = nil
        default:
            break
        }
    }

    /// The ⇄ button: the other direction, at the stop nearest to the
    /// current one.
    func flipDirection() async {
        direction = 1 - direction
        headsign = nil
        requestedStop = nil
        let fromStop = highlightedStop
        await load(nearestTo: fromStop?.coordinate ?? centre)
        startRefreshing()
    }

    // MARK: - Loading

    private func load(nearestTo point: CLLocationCoordinate2D? = nil) async {
        isLoading = true
        generation += 1
        let generation = generation
        let route = route
        let direction = direction
        async let loadedStops = staticData.stops(for: route, direction: direction)
        async let loadedShapes = shapes.shapes(for: route.name)
        async let listed = staticData.routes(for: mode)
        let (routeStops, routeShapes, routes) = await (loadedStops, loadedShapes, listed)
        guard generation == self.generation else { return }

        stops = routeStops
        shape = routeShapes.first { $0.direction == direction }
        highlightedStop = Self.pickStop(requested: requestedStop, in: routeStops, nearestTo: point ?? requestedStop?.coordinate ?? centre)
        if headsign == nil {
            let fromList = routes.first { $0.name == route.name && $0.agencyId == route.agencyId }?.headsign(for: direction)
            headsign = fromList ?? headsignLookup["\(route.name)_\(direction)"]
        }
        isLoading = false
    }

    /// The stop to highlight: the requested stop when the list has it, else
    /// the requested stop itself (the timeline's pin is newer than a bundled
    /// stop list), else the stop nearest to `point`.
    nonisolated static func pickStop(requested: FocusStop?, in stops: [RouteStop], nearestTo point: CLLocationCoordinate2D) -> RouteStop? {
        if let requested {
            return stops.first { $0.id == requested.id } ?? requested.routeStop
        }
        return stops.min { lhs, rhs in
            distance(from: point, to: lhs) < distance(from: point, to: rhs)
        }
    }

    private nonisolated static func distance(from point: CLLocationCoordinate2D, to stop: RouteStop) -> Double {
        GeoUtils.straightLineDistance(lat1: point.latitude, lon1: point.longitude, lat2: stop.lat, lon2: stop.lon)
    }

    /// A region fitting the stop and the shape points within `focusRadius`
    /// of it, padded 30%; just the stop (0.008° span) when there is no line.
    nonisolated static func focusRegion(stop: RouteStop, shape: RouteShape?) -> MKCoordinateRegion {
        let near = (shape?.clCoordinates ?? []).filter {
            GeoUtils.straightLineDistance(lat1: stop.lat, lon1: stop.lon, lat2: $0.latitude, lon2: $0.longitude) <= focusRadius
        }
        guard !near.isEmpty else {
            return MKCoordinateRegion(center: stop.coordinate, span: MKCoordinateSpan(latitudeDelta: 0.008, longitudeDelta: 0.008))
        }
        let points = near + [stop.coordinate]
        let minLat = points.map(\.latitude).min()!, maxLat = points.map(\.latitude).max()!
        let minLon = points.map(\.longitude).min()!, maxLon = points.map(\.longitude).max()!
        return MKCoordinateRegion(
            center: CLLocationCoordinate2D(latitude: (minLat + maxLat) / 2, longitude: (minLon + maxLon) / 2),
            span: MKCoordinateSpan(
                latitudeDelta: max(0.005, (maxLat - minLat) * 1.3),
                longitudeDelta: max(0.005, (maxLon - minLon) * 1.3)
            )
        )
    }

    // MARK: - Refreshing

    private func startRefreshing() {
        refreshTask?.cancel()
        guard isActive else { return }
        let clock = clock
        refreshTask = Task { [weak self] in
            while !Task.isCancelled {
                await self?.refresh()
                guard (try? await clock.sleep(for: Self.refreshInterval)) != nil else { return }
            }
        }
    }

    /// Fetches the vehicles on this direction and the arrivals at the
    /// highlighted stop. Failures keep the last answer and set a message.
    func refresh() async {
        let generation = generation
        let (source, route, mode, direction, stop) = (source, route, mode, direction, highlightedStop)
        do {
            // The Luas has no live positions in TFI's feed, so do not ask.
            async let fetchedVehicles: [Vehicle] = {
                guard mode != .luas else { return [] }
                return try await source.fetchVehicles(mode: mode)
            }()
            async let fetchedArrivals: [StopArrival] = {
                guard let stop else { return [] }
                if mode == .luas {
                    return try await source.fetchLuasArrivals(stopId: stop.id, direction: direction, routeName: route.name)
                }
                return try await source.fetchArrivals(stopId: stop.id, route: route, direction: direction)
            }()
            let (allVehicles, nextArrivals) = try await (fetchedVehicles, fetchedArrivals)
            guard generation == self.generation else { return }
            vehicles = allVehicles.filter { route.matches(routeId: $0.routeId, agencyId: $0.agencyId) && $0.direction == direction }
            arrivals = nextArrivals
            lastUpdated = clock.now()
            errorMessage = nil
        } catch {
            guard generation == self.generation, !ServerErrorText.isCancellation(error) else { return }
            Self.logger.error("Focus refresh failed: \(error.localizedDescription)")
            errorMessage = ServerErrorText.message(for: error)
        }
    }
}
```

```swift
/// What HomeModel needs from the location service, so tests can fake it
/// without a CLLocationManager.
@MainActor
protocol LocationProviding: AnyObject {
    var userLocation: CLLocationCoordinate2D? { get }
    var authorizationStatus: CLAuthorizationStatus { get }
    var authorizationStatusPublisher: AnyPublisher<CLAuthorizationStatus, Never> { get }
    /// Asks for permission if needed, then returns the location once; nil
    /// when denied or no fix is available.
    func requestLocation() async -> CLLocationCoordinate2D?
```

```swift
import Combine
import CoreLocation
import Foundation
import MapKit
import SwiftUI

/// The home screen: which panel is showing, the mode, the map camera and
/// the user's location. Rows, search results and route focus live in the
/// child models; this one wires the transitions between them.
///
/// The timeline follows the map: the view reports where the camera
/// settled and `NearbyModel` fetches for it. Opening route focus pauses
/// the timeline and moves the camera to the route; closing it restores the
/// camera, which (being within 50 m of the last fetch) leaves the rows and
/// scroll position intact.
@MainActor
final class HomeModel: ObservableObject {
    enum Screen: Equatable {
        case timeline
        case search
        case focus(RouteFocusTarget)
    }

    enum LocationState: Equatable {
        /// Authorisation not determined: the priming card is showing.
        case undecided
        /// Following the user's location.
        case located
        /// Not now, denied or restricted: the map started at O'Connell Bridge.
        case cityCentre
        /// Authorisation was taken away while running; the map stays put.
        case off
    }

    typealias FocusFactory = @MainActor (RouteFocusTarget, CLLocationCoordinate2D, CLLocationCoordinate2D?) -> RouteFocusModel

    /// O'Connell Bridge.
    static let cityCentre = CLLocationCoordinate2D(latitude: 53.3472, longitude: -6.2592)
    static let homeSpan = MKCoordinateSpan(latitudeDelta: 0.012, longitudeDelta: 0.012)

    @Published private(set) var screen: Screen = .timeline
    @Published private(set) var mode: TransportMode = .bus
    @Published var cameraPosition: MapCameraPosition = .region(MKCoordinateRegion(center: HomeModel.cityCentre, span: HomeModel.homeSpan))
    @Published private(set) var locationState: LocationState = .undecided
    /// Where the camera last settled.
    @Published private(set) var centre: CLLocationCoordinate2D = HomeModel.cityCentre
    @Published private(set) var focus: RouteFocusModel?
    /// A row id the timeline should scroll to (a pin was tapped); the view
    /// clears it after scrolling.
    @Published var scrollToRow: String?

    let nearby: NearbyModel
    let search: SearchModel

    private let location: LocationProviding
    private let makeFocus: FocusFactory
    private var timelineCamera: MapCameraPosition?
    private var hasAppeared = false
    private var cancellables: Set<AnyCancellable> = []

    init(nearby: NearbyModel, search: SearchModel, location: LocationProviding, makeFocus: @escaping FocusFactory) {
        self.nearby = nearby
        self.search = search
        self.location = location
        self.makeFocus = makeFocus
        // Views observe this model; forward the children's changes.
        nearby.objectWillChange.sink { [weak self] _ in self?.objectWillChange.send() }.store(in: &cancellables)
        search.objectWillChange.sink { [weak self] _ in self?.objectWillChange.send() }.store(in: &cancellables)
        location.authorizationStatusPublisher
            .dropFirst()
            .sink { [weak self] status in self?.authorizationChanged(to: status) }
            .store(in: &cancellables)
    }

    var showsPrimingCard: Bool { locationState == .undecided && screen == .timeline }

    var locationNote: String? {
        switch locationState {
        case .cityCentre: return "Showing city centre"
        case .off: return "Location is off"
        case .undecided, .located: return nil
        }
    }

    /// Whether the near-me button can do anything.
    var canLocate: Bool {
        switch location.authorizationStatus {
        case .notDetermined, .authorizedWhenInUse, .authorizedAlways: return true
        default: return false
        }
    }

    var userLocation: CLLocationCoordinate2D? { location.userLocation }

    // MARK: - Lifecycle

    func onAppear() {
        guard !hasAppeared else { return }
        hasAppeared = true
        switch location.authorizationStatus {
        case .notDetermined:
            locationState = .undecided
        case .authorizedWhenInUse, .authorizedAlways:
            Task { await locate() }
        default:
            locationState = .cityCentre
        }
    }

    func scenePhaseChanged(to phase: ScenePhase) {
        focus?.scenePhaseChanged(to: phase)
        if case .focus = screen {
            return // The timeline is paused while focus is open.
        }
        nearby.scenePhaseChanged(to: phase)
    }

    /// The map camera stopped moving. In route focus the camera follows the
    /// route, not the user, so the timeline is left alone.
    func mapCameraSettled(centre: CLLocationCoordinate2D) {
        self.centre = centre
        if case .focus = screen { return }
        nearby.cameraSettled(at: centre)
    }

    // MARK: - Location

    /// "Allow location" on the priming card: shows the iOS prompt.
    func allowLocation() {
        Task { await locate() }
    }

    /// "Not now" on the priming card.
    func notNow() {
        locationState = .cityCentre
        cameraPosition = .region(MKCoordinateRegion(center: Self.cityCentre, span: Self.homeSpan))
    }

    /// The near-me button: back to the user's location (asking first if
    /// authorisation is still undecided).
    func goNearMe() {
        Task { await locate() }
    }

    private func locate() async {
        if let coordinate = await location.requestLocation() {
            locationState = .located
            cameraPosition = .region(MKCoordinateRegion(center: coordinate, span: Self.homeSpan))
        } else if locationState != .located {
            locationState = .cityCentre
            cameraPosition = .region(MKCoordinateRegion(center: Self.cityCentre, span: Self.homeSpan))
        }
    }

    private func authorizationChanged(to status: CLAuthorizationStatus) {
        switch status {
        case .denied, .restricted:
            // Keep the map where it is; the timeline still follows the map.
            locationState = locationState == .undecided ? .cityCentre : .off
        default:
            break
        }
    }

    // MARK: - Modes

    func selectMode(_ mode: TransportMode) {
        guard mode != self.mode else { return }
        self.mode = mode
        nearby.setMode(mode)
        search.setMode(mode)
    }

    // MARK: - Search

    func beginSearch() {
        guard screen == .timeline else { return }
        screen = .search
    }

    func cancelSearch() {
        search.clear()
        screen = .timeline
    }

    /// A stop result: fly there; the timeline then follows the map.
    func selectStopResult(_ stop: StopSearchResult) {
        fly(to: stop.coordinate)
    }

    func selectPlaceResult(_ place: PlaceResult) {
        fly(to: place.coordinate)
    }

    /// A route result: focus, outbound, at the stop nearest the map centre.
    func selectRouteResult(_ route: RouteSearchResult) {
        search.clear()
        openFocus(RouteFocusTarget(route: route.ref, mode: route.mode, direction: 0, headsign: route.headsign(for: 0), stop: nil))
    }

    private func fly(to coordinate: CLLocationCoordinate2D) {
        search.clear()
        screen = .timeline
        cameraPosition = .region(MKCoordinateRegion(center: coordinate, span: Self.homeSpan))
    }

    // MARK: - Route focus

    /// A timeline row: focus on its route, direction and stop.
    func openFocus(departure: NearbyDeparture) {
        let stop = nearby.result?.stop(withId: departure.stopId).map(FocusStop.init)
        openFocus(RouteFocusTarget(route: departure.routeRef, mode: mode, direction: departure.direction, headsign: departure.headsign, stop: stop))
    }

    private func openFocus(_ target: RouteFocusTarget) {
        timelineCamera = cameraPosition
        nearby.pause()
        let model = makeFocus(target, centre, location.userLocation)
        model.objectWillChange.sink { [weak self] _ in self?.objectWillChange.send() }.store(in: &cancellables)
        focus = model
        screen = .focus(target)
        Task { [weak self] in
            await model.start()
            guard let self, self.focus === model, let region = model.focusRegion else { return }
            self.cameraPosition = .region(region)
        }
    }

    /// Back chevron or swipe down: the timeline, as it was.
    func closeFocus() {
        focus?.stop()
        focus = nil
        screen = .timeline
        if let timelineCamera {
            cameraPosition = timelineCamera
        }
        nearby.resume()
    }

    // MARK: - Pins

    /// Tapping a pin scrolls the timeline to that stop's first row.
    func selectPin(stopId: String) {
        guard let row = nearby.firstRow(atStop: stopId) else { return }
        scrollToRow = row.id
    }
}
```

```swift
                                        now: context.date
```

```swift
/// Canned answers for the UI smoke test (launched with TURAS_UITEST_STUB=1),
/// so it runs the same on CI without a server. Times are built from the
/// current date so rows look live. Only the paths the home screen uses are

enum UITestStubServer {
    static func session() -> URLSession {
        let configuration = URLSessionConfiguration.ephemeral
        configuration.protocolClasses = [StubProtocol.self]
        return URLSession(configuration: configuration)
    }

    static func body(for url: URL) -> (status: Int, json: String) {
        let items = URLComponents(url: url, resolvingAgainstBaseURL: false)?.queryItems ?? []
        func value(_ name: String) -> String? { items.first { $0.name == name }?.value }
        switch url.path {
        case "/api/v1/nearby":
            return (200, nearby(mode: value("mode") ?? "bus"))
        case "/api/v1/search":
            return (200, search(query: value("q") ?? ""))
        case "/api/v1/stops":
            return (200, stops(route: value("route") ?? "", direction: value("direction") ?? "0"))
        case "/api/v1/vehicles":
            return (200, vehicles(mode: value("mode") ?? "bus"))
        case "/api/v1/arrivals", "/api/v1/luas/arrivals":
            return (200, arrivals(route: value("route") ?? "39A"))
        case "/api/v1/routes":
            return (200, "[]")
        default:
            return (404, #"{"error": "not stubbed"}"#)
        }
    }

        let formatter = ISO8601DateFormatter()
        formatter.formatOptions = [.withInternetDateTime]
        return formatter
```
